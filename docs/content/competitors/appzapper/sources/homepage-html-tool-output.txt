            pad: 24,                 // px above and below the tallest column
            colGap: 30,              // px between columns
            colStart: [0, 0, 0, 0],  // px nudge per column off true centre (none: columns share a midline)
            swapEvery: [3500, 6000], // ms between swaps (one bubble at a time)
            pause: [350, 800],       // ms a slot sits empty between bubbles
            rotate: 3,               // max tilt, degrees either way
            jitterX: 10,             // max sideways nudge, px
        };

        const rand = (a, b) => a + Math.random() * (b - a);
        let cloudSlots = [];   // every slot, in column order
        let cloudColumns = []; // slots grouped by column
        let cloudQueue = [];
        let cloudCols = 0, cloudPerCol = 0;
        let cloudVisible = true;

        // Shuffled pass through the pool; never hands back a quote that is
        // currently on screen. On narrow two-column layouts only the shorter
        // quotes are used, so nothing has to wrap into a tower.
        let cloudMaxChars = Infinity;

        function nextQuote() {
            if (!cloudQueue.length) {
                cloudQueue = QUOTES.map((_, i) => i).filter((i) => QUOTES[i][0].length <= cloudMaxChars);
                for (let i = cloudQueue.length - 1; i > 0; i--) {
                    const j = Math.floor(Math.random() * (i + 1));
                    [cloudQueue[i], cloudQueue[j]] = [cloudQueue[j], cloudQueue[i]];
                }
            }
            const onScreen = new Set(cloudSlots.map((s) => s.quoteIndex));
            let idx = cloudQueue.findIndex((q) => !onScreen.has(q));
            if (idx < 0) idx = 0;
            return cloudQueue.splice(idx, 1)[0];
        }

        function makeBubble(qi, side) {
            const [text, who] = QUOTES[qi];
            const el = document.createElement('blockquote');
            el.className = 'quote ' + side + (text.length > 105 ? ' long' : '');
            const inner = document.createElement('div');
            inner.className = 'quote-inner';
            inner.append(`"${text}"`);
            const span = document.createElement('span');
            span.className = 'who';
            span.textContent = `— ${who}`;
            inner.appendChild(span);
            el.appendChild(inner);
            return el;
        }

        // Stack a column's bubbles centred vertically in the cloud (plus a
        // small per-column nudge so rows never align); slots animate to
        // their new tops (CSS transition), so a taller or shorter arrival
        // smoothly spreads the column out or draws it in around the middle.
        function columnHeight(c) {
            const filled = cloudColumns[c].filter((s) => s.h);
            return filled.reduce((sum, s) => sum + s.h, 0) + CLOUD.gap * Math.max(0, filled.length - 1);
        }

        function layoutColumn(c, areaH) {
            let y = (areaH - columnHeight(c)) / 2 + CLOUD.colStart[c % CLOUD.colStart.length];
            for (const slot of cloudColumns[c]) {
                slot.el.style.top = y.toFixed(1) + 'px';
                y += slot.h + (slot.h ? CLOUD.gap : 0);
            }
        }

        // Size the area to the tallest column (plus breathing room) and
        // centre every column in it, so no stack can ever run into the copy
        // above or the wave below; the height eases along with the slots.
        function fitCloud() {
            const areaH = Math.max(...cloudColumns.map((_, c) => columnHeight(c))) + CLOUD.pad * 2;
            cloud.parentElement.style.height = areaH + 'px';
            cloudColumns.forEach((_, c) => layoutColumn(c, areaH));
        }

        // Put the next quote in a slot, measured but hidden, without
        // touching the layout yet.
        function placeInSlot(slot) {
            const qi = nextQuote();
            slot.quoteIndex = qi;
            const el = makeBubble(qi, Math.random() < 0.5 ? 'left' : 'right');
            el.classList.add('pending');
            el.style.setProperty('--rot', rand(-CLOUD.rotate, CLOUD.rotate).toFixed(1) + 'deg');
            el.style.setProperty('--jx', rand(-CLOUD.jitterX, CLOUD.jitterX).toFixed(1) + 'px');
            const inner = el.firstElementChild;
            inner.style.animationDuration = rand(8, 12).toFixed(2) + 's';
            inner.style.animationDelay = (-rand(0, 12)).toFixed(2) + 's';
            slot.el.replaceChildren(el);
            slot.bubble = el;
            slot.h = el.offsetHeight; // unaffected by the pop-in transform
        }

        function revealSlot(slot) {
            if (slot.bubble) slot.bubble.classList.remove('pending');
        }

        function showInSlot(slot) {
            placeInSlot(slot);
            fitCloud();
            revealSlot(slot);
        }

        // Pop this slot's bubble out, then pop the next quote in after a beat.
        // The slot keeps its old height while empty so the column only
        // reflows once the newcomer's size is known.
        function swapSlot(slot) {
            const el = slot.bubble;
            if (!el) return;
            el.classList.add('out');
            slot.quoteIndex = -1;
            slot.timer = setTimeout(() => {
                if (slot.bubble === el) { el.remove(); slot.bubble = null; }
                showInSlot(slot);
            }, 500 + rand(...CLOUD.pause));
        }

        // One swap at a time, on a lazy random cadence; never the same slot
        // twice in a row, and it waits while the section is off screen.
        let swapTimer = 0;
        let lastSwapped = null;
        function swapOne() {
            if (!cloudVisible) { swapTimer = setTimeout(swapOne, 1000); return; }
            const candidates = cloudSlots.filter((s) => s.bubble && s !== lastSwapped);
            if (candidates.length) {
                const slot = candidates[Math.floor(Math.random() * candidates.length)];
                lastSwapped = slot;
                swapSlot(slot);
            }
            swapTimer = setTimeout(swapOne, rand(...CLOUD.swapEvery));
        }

        // Columns of stacked bubbles, each column's stack starting at a
        // slightly different height so rows never line up. Rebuilt only
        // when the column count changes; a plain resize just reflows.
        function buildCloud() {
            const w = cloud.clientWidth;
            // phones get a single column; otherwise never fewer than two
            const narrow = window.innerWidth < 480; // phones only
            const cols = narrow ? 1 : Math.max(2, Math.min(CLOUD.maxCols, Math.floor(w / CLOUD.cellWidth)));
            const perCol = cols === 1 ? 6 : CLOUD.perCol; // one column stacks six, others five
            const maxChars = Infinity; // full-width phone bubbles fit every quote
            if (cols === cloudCols && perCol === cloudPerCol && maxChars === cloudMaxChars) {
                reflowCloud();
                return;
            }
            cloudCols = cols;
            cloudPerCol = perCol;
            cloudMaxChars = maxChars;
            cloudSlots.forEach((s) => clearTimeout(s.timer));
            cloud.replaceChildren();
            cloudSlots = [];
            cloudColumns = [];
            const colW = `calc((100% - ${CLOUD.colGap * (cols - 1)}px) / ${cols})`;
            for (let c = 0; c < cols; c++) {
                const column = [];
                for (let i = 0; i < perCol; i++) {
                    const el = document.createElement('div');
                    el.className = 'quote-slot';
                    el.style.width = colW;
                    el.style.left = `calc(${c} * (${colW} + ${CLOUD.colGap}px))`;
                    el.style.top = '0px';
                    cloud.appendChild(el);
                    const slot = { el, col: c, h: 0, bubble: null, timer: 0, quoteIndex: -1 };
                    column.push(slot);
                    cloudSlots.push(slot);
                }
                cloudColumns.push(column);
            }
            cloudQueue = [];
            // Place and measure every bubble first, lay the columns out at
            // their final positions with no sliding, then reveal top to
            // bottom (columns interleaved) so nothing shuffles as it appears.
            cloud.classList.add('no-anim');
            cloudSlots.forEach(placeInSlot);
            fitCloud();
            requestAnimationFrame(() => requestAnimationFrame(() => cloud.classList.remove('no-anim')));
            // then reveal the whole wall in one go
            cloudSlots.forEach(revealSlot);
            clearTimeout(swapTimer);
            lastSwapped = null;
            if (!reducedMotion) { // static wall under reduced motion
                swapTimer = setTimeout(swapOne, 1500 + rand(...CLOUD.swapEvery));
            }
        }

        // Widths changed, so heights may have too: re-measure and restack
        // in place, without the slide.
        function reflowCloud() {
            cloud.classList.add('no-anim');
            for (const slot of cloudSlots) if (slot.bubble) slot.h = slot.bubble.offsetHeight;
            fitCloud();
            requestAnimationFrame(() => requestAnimationFrame(() => cloud.classList.remove('no-anim')));
        }

        buildCloud();
        window.addEventListener('resize', buildCloud, { passive: true });
        new IntersectionObserver((entries) => {
            cloudVisible = entries.some((e) => e.isIntersecting);
        }).observe(cloud);

        // ------------------------------------------------------------------
        // 3D viewer: the gun holds its spot and pivots gently with scroll —
        // yaw and pitch together, starting muzzle-left like in the app.
        // ------------------------------------------------------------------
        const mv = document.getElementById('zapper3d');

        if (!reducedMotion) {
            // Let the user spin the gun while held; the moment they release,
            // retarget the scroll pose and model-viewer glides it home.
            let draggingGun = false;
            mv.addEventListener('pointerdown', () => { draggingGun = true; });
            window.addEventListener('pointerup', () => {
                if (draggingGun) { draggingGun = false; poseFromScroll(); }
            });
            window.addEventListener('pointercancel', () => {
                if (draggingGun) { draggingGun = false; poseFromScroll(); }
            });

            // Scroll animates the MODEL's orientation through three keyframes
            // (roll, pitch, yaw); the camera stays parked dead ahead. The end
            // keyframes are reached exactly when the stage is fully offscreen.
            const ORIENT_BELOW = [-8, 43, 37];   // fully offscreen below
            const ORIENT_CENTER = [-26, 29, 71]; // hand-tuned center-on pose
            const ORIENT_ABOVE = [-8, 4, 99];    // fully offscreen above

            function lerpOrient(a, b, t) {
                return a.map((v, i) => v + (b[i] - v) * t);
            }

            // Scroll only retargets; the pin loop eases toward the target so
            // notchy wheel scrolling still reads as one fluid motion.
            let targetOrient = ORIENT_CENTER.slice();
            let currentOrient = ORIENT_CENTER.slice();
            let poseQueued = false;
            function poseFromScroll() {
                if (draggingGun) return;
                const r = mv.getBoundingClientRect();
                // 0 with the stage centered; ±1 exactly when it has fully
                // left the viewport (above positive, below negative).
                const travel = window.innerHeight / 2 + r.height / 2;
                const p = Math.max(-1, Math.min(1,
                    ((window.innerHeight / 2) - (r.top + r.height / 2)) / travel));
                targetOrient = p < 0
                    ? lerpOrient(ORIENT_CENTER, ORIENT_BELOW, -p)
                    : lerpOrient(ORIENT_CENTER, ORIENT_ABOVE, p);
                // home the camera too (it only moves if the user dragged it)
                mv.cameraOrbit = '0deg 90deg auto';
            }
            window.addEventListener('scroll', () => {
                if (poseQueued) return;
                poseQueued = true;
                requestAnimationFrame(() => { poseQueued = false; poseFromScroll(); });
            }, { passive: true });
            window.addEventListener('resize', poseFromScroll, { passive: true });
            poseFromScroll();

            // Pin the callout cards to points on the model: each card drifts
            // by half of its anchor's on-screen movement from the rest pose.
            const stage = document.querySelector('.zapper-stage');
            const compactCallouts = window.matchMedia('(max-width: 900px)');
            const cardAnchors = [];
            for (let i = 1; i <= 6; i++) {
                const card = document.querySelector('.callout.c' + i);
                const anchor = stage.querySelector('.anchor[data-anchor="c' + i + '"]');
                if (!card || !anchor) continue;
                const tether = document.createElement('div');
                tether.className = 'tether';
                tether.setAttribute('aria-hidden', 'true');
                stage.appendChild(tether);
                // the authored data-position is in the model's unrotated space
                const base = anchor.dataset.position.split(/\s+/).map(parseFloat);
                cardAnchors.push({ card, anchor, tether, base, baseFx: null, baseFy: null, home: null });
            }

            // Dots, tethers, and card drift all derive from ONE analytic
            // projection computed from the same orientation value we just
            // applied — never read back from model-viewer's DOM, which updates
            // a frame late and reads as jitter while scrolling.
            function anchorScreenPoints(orient, restCamera) {
                if (!mv.loaded || typeof mv.getCameraOrbit !== 'function') return null;
                const box = mv.getBoundingClientRect();
                if (!box.width || !box.height) return null;
                const live = mv.getCameraOrbit();
                const theta = restCamera ? 0 : live.theta;
                const phi = restCamera ? Math.PI / 2 : live.phi;
                const radius = live.radius;
                const tgt = mv.getCameraTarget();
                const halfFovY = mv.getFieldOfView() * Math.PI / 360;

                // camera on the orbit sphere, looking at the target
                const sp = Math.sin(phi), cp = Math.cos(phi);
                const st = Math.sin(theta), ct = Math.cos(theta);
                const cam = [tgt.x + radius * sp * st, tgt.y + radius * cp, tgt.z + radius * sp * ct];
                let f = [tgt.x - cam[0], tgt.y - cam[1], tgt.z - cam[2]];
                const fl = Math.hypot(f[0], f[1], f[2]);
                f = [f[0] / fl, f[1] / fl, f[2] / fl];
                let r = [-f[2], 0, f[0]];
                const rl = Math.hypot(r[0], r[1], r[2]) || 1;
                r = [r[0] / rl, r[1] / rl, r[2] / rl];
                const u = [
                    r[1] * f[2] - r[2] * f[1],
                    r[2] * f[0] - r[0] * f[2],
                    r[0] * f[1] - r[1] * f[0],
                ];
                const tanY = Math.tan(halfFovY);
                const tanX = tanY * box.width / box.height;

                // model rotation — same YXZ Euler model-viewer applies
                const [rollD, pitchD, yawD] = orient;
                const rr = rollD * Math.PI / 180;
                const pp = pitchD * Math.PI / 180;
                const yy = yawD * Math.PI / 180;
                const cy = Math.cos(yy), sy = Math.sin(yy);
                const cx = Math.cos(pp), sx = Math.sin(pp);
                const cz = Math.cos(rr), sz = Math.sin(rr);
                const m = [
                    [cy * cz + sy * sx * sz, -cy * sz + sy * sx * cz, sy * cx],
                    [cx * sz, cx * cz, -sx],
                    [-sy * cz + cy * sx * sz, sy * sz + cy * sx * cz, cy * cx],
                ];

                return cardAnchors.map((ca) => {
                    const [x, y, z] = ca.base;
                    const w = [
                        m[0][0] * x + m[0][1] * y + m[0][2] * z - cam[0],
                        m[1][0] * x + m[1][1] * y + m[1][2] * z - cam[1],
                        m[2][0] * x + m[2][1] * y + m[2][2] * z - cam[2],
                    ];
                    const zc = w[0] * f[0] + w[1] * f[1] + w[2] * f[2];
                    const xc = w[0] * r[0] + w[1] * r[1] + w[2] * r[2];
                    const yc = w[0] * u[0] + w[1] * u[1] + w[2] * u[2];
                    return {
                        x: box.left + (0.5 + xc / zc / (2 * tanX)) * box.width,
                        y: box.top + (0.5 - yc / zc / (2 * tanY)) * box.height,
                    };
                });
            }

            function captureBaselines() {
                // rest pose is computed analytically, so baselines don't care
                // what the current scroll/drag state is
                const pts = anchorScreenPoints(ORIENT_CENTER, true);
                if (!pts) return;
                const box = mv.getBoundingClientRect();
                const stageR = stage.getBoundingClientRect();
                cardAnchors.forEach((ca, i) => {
                    ca.baseFx = (pts[i].x - box.left) / box.width;
                    ca.baseFy = (pts[i].y - box.top) / box.height;
                    ca.card.style.transform = '';
                    const cr = ca.card.getBoundingClientRect();
                    ca.home = {
                        x: cr.left - stageR.left,
                        y: cr.top - stageR.top,
                        w: cr.width,
                        h: cr.height,
                    };
                });
            }

            mv.addEventListener('load', () => {
                setTimeout(() => { captureBaselines(); poseFromScroll(); }, 100);
            });
            window.addEventListener('resize', () => {
                if (mv.loaded) captureBaselines();
            }, { passive: true });

            let lastPoseTick = performance.now();
            (function pinTick(now) {
                requestAnimationFrame(pinTick);
                // ease the orientation toward the scroll target
                const dt = Math.min(0.1, ((now || performance.now()) - lastPoseTick) / 1000);
                lastPoseTick = now || performance.now();
                const ease = 1 - Math.pow(0.002, dt);
                let moved = false;
                for (let i = 0; i < 3; i++) {
                    const d = targetOrient[i] - currentOrient[i];
                    if (Math.abs(d) > 0.005) {
                        currentOrient[i] += d * ease;
                        moved = true;
                    }
                }
                if (moved) {
                    mv.orientation = `${currentOrient[0].toFixed(2)}deg ${currentOrient[1].toFixed(2)}deg ${currentOrient[2].toFixed(2)}deg`;
                }

                const pts = anchorScreenPoints(currentOrient, false);
                if (!pts) return;
                const box = mv.getBoundingClientRect();
                const stageR = stage.getBoundingClientRect();
                cardAnchors.forEach((ca, i) => {
                    if (ca.card.offsetParent === null) {
                        // hidden at this breakpoint (compact drops c3/c4)
                        ca.tether.style.display = 'none';
                        return;
                    }
                    const ax = pts[i].x;
                    const ay = pts[i].y;
                    ca.anchor.style.left = (ax - stageR.left).toFixed(1) + 'px';
                    ca.anchor.style.top = (ay - stageR.top).toFixed(1) + 'px';
                    ca.anchor.style.visibility = 'visible';
                    if (ca.baseFx === null) return;

                    // drift the card by half the anchor's movement, but never
                    // push it out of the stage
                    let dx = (ax - box.left - ca.baseFx * box.width) * 0.5;
                    let dy = (ay - box.top - ca.baseFy * box.height) * 0.5;
                    dx = Math.max(4 - ca.home.x, Math.min(stageR.width - ca.home.w - 4 - ca.home.x, dx));
                    dy = Math.max(2 - ca.home.y, Math.min(stageR.height - ca.home.h - 2 - ca.home.y, dy));
                    ca.card.style.transform = `translate(${dx.toFixed(2)}px, ${dy.toFixed(2)}px)`;

                    // line from the card's gun-facing edge straight to the
                    // marker — side edges on desktop, top/bottom when compact
                    const cr = ca.card.getBoundingClientRect();
                    let sx, sy;
                    if (compactCallouts.matches && !ca.card.classList.contains('c3')) {
                        sx = cr.left + cr.width / 2 - stageR.left;
                        sy = (ay < cr.top + cr.height / 2 ? cr.top : cr.bottom) - stageR.top;
                    } else {
                        const fromRight = ax >= cr.left + cr.width / 2;
                        sx = (fromRight ? cr.right : cr.left) - stageR.left;
                        sy = cr.top + cr.height / 2 - stageR.top;
                    }
                    const ex = ax - stageR.left;
                    const ey = ay - stageR.top;
                    const len = Math.hypot(ex - sx, ey - sy);
                    ca.tether.style.display = '';
                    ca.tether.style.left = sx + 'px';
                    ca.tether.style.top = sy + 'px';
                    ca.tether.style.width = len.toFixed(1) + 'px';
                    ca.tether.style.transform = `rotate(${Math.atan2(ey - sy, ex - sx)}rad)`;
                });
            })();

        }
